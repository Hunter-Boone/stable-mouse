cmake_minimum_required(VERSION 3.22)
project(StableMouse VERSION 0.1.6 LANGUAGES CXX)
set(STABLE_MOUSE_RELEASE_VERSION "" CACHE STRING "Full release version, matching the GitHub tag without v; empty uses the project version")
if(STABLE_MOUSE_RELEASE_VERSION STREQUAL "")
  if(DEFINED ENV{STABLE_MOUSE_RELEASE_VERSION} AND NOT "$ENV{STABLE_MOUSE_RELEASE_VERSION}" STREQUAL "")
    set(STABLE_MOUSE_RELEASE_VERSION "$ENV{STABLE_MOUSE_RELEASE_VERSION}")
  else()
    set(STABLE_MOUSE_RELEASE_VERSION "${PROJECT_VERSION}")
  endif()
endif()
string(REPLACE "." "\\." STABLE_MOUSE_VERSION_PATTERN "${PROJECT_VERSION}")
if(NOT STABLE_MOUSE_RELEASE_VERSION MATCHES "^${STABLE_MOUSE_VERSION_PATTERN}(-[0-9A-Za-z.-]+)?(\\+[0-9A-Za-z.-]+)?$")
  message(FATAL_ERROR "STABLE_MOUSE_RELEASE_VERSION must start with project version ${PROJECT_VERSION} and use a semantic version suffix")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
include(CTest)
add_library(stabilizer INTERFACE)
target_include_directories(stabilizer INTERFACE src)
if(BUILD_TESTING)
  add_executable(filter_tests tests/filter_tests.cpp)
  target_link_libraries(filter_tests PRIVATE stabilizer)
  add_test(NAME filter COMMAND filter_tests)
  add_executable(motion_benchmark tests/motion_benchmark.cpp)
  target_link_libraries(motion_benchmark PRIVATE stabilizer)
  add_test(NAME motion_scenarios COMMAND motion_benchmark)
  add_executable(center_tests tests/center_tests.cpp)
  target_link_libraries(center_tests PRIVATE stabilizer)
  add_test(NAME center COMMAND center_tests)
  add_executable(variance_tests tests/variance_tests.cpp)
  target_link_libraries(variance_tests PRIVATE stabilizer)
  add_test(NAME variance COMMAND variance_tests)
  add_executable(always_tests tests/always_tests.cpp)
  target_link_libraries(always_tests PRIVATE stabilizer)
  add_test(NAME always_center COMMAND always_tests)
endif()
option(STABLE_MOUSE_CORE_ONLY "Build only the portable filter tests" OFF)
if(STABLE_MOUSE_CORE_ONLY)
  return()
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network Test)
qt_standard_project_setup()
add_library(stable_mouse_updater STATIC src/updater.cpp src/updater.h src/update_widget.cpp src/update_widget.h)
target_link_libraries(stable_mouse_updater PUBLIC Qt6::Widgets Qt6::Network)
target_include_directories(stable_mouse_updater PUBLIC src)
if(WIN32)
  target_link_libraries(stable_mouse_updater PRIVATE shell32)
endif()
qt_add_resources(STABLE_MOUSE_UI_RESOURCES src/ui.qrc)
qt_add_executable(stable-mouse WIN32 MACOSX_BUNDLE
  ${STABLE_MOUSE_UI_RESOURCES}
  src/main.cpp src/window.cpp src/window.h src/practice.cpp src/practice.h
  src/startup.cpp src/startup.h src/backend.h)
target_link_libraries(stable-mouse PRIVATE stabilizer stable_mouse_updater)
target_compile_definitions(stable-mouse PRIVATE STABLE_MOUSE_VERSION="${STABLE_MOUSE_RELEASE_VERSION}")
if(WIN32)
  enable_language(RC)
  configure_file(packaging/windows-icon.rc.in windows-icon.rc @ONLY)
  set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/windows-icon.rc" PROPERTIES
    OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/packaging/stable-mouse.ico")
  target_sources(stable-mouse PRIVATE src/platform/windows.cpp "${CMAKE_CURRENT_BINARY_DIR}/windows-icon.rc")
  target_compile_definitions(stable-mouse PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_link_libraries(stable-mouse PRIVATE user32)
elseif(APPLE)
  target_sources(stable-mouse PRIVATE src/platform/macos.cpp)
  target_link_libraries(stable-mouse PRIVATE "-framework ApplicationServices" "-framework Carbon")
  set_target_properties(stable-mouse PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Stable Mouse"
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.stablemouse.StableMouse"
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(EVDEV REQUIRED IMPORTED_TARGET libevdev)
  target_sources(stable-mouse PRIVATE src/platform/linux.cpp)
  add_executable(stable-mouse-input src/platform/linux_helper.cpp)
  target_link_libraries(stable-mouse-input PRIVATE stabilizer PkgConfig::EVDEV)
  install(TARGETS stable-mouse-input RUNTIME DESTINATION libexec)
  install(FILES packaging/org.stablemouse.StableMouse.desktop DESTINATION share/applications)
  install(FILES packaging/stable-mouse.svg DESTINATION share/icons/hicolor/scalable/apps)
endif()
install(TARGETS stable-mouse BUNDLE DESTINATION . RUNTIME DESTINATION bin)
if(APPLE)
  set(STABLE_MOUSE_DOC_DIR "stable-mouse.app/Contents/Resources/doc")
else()
  set(STABLE_MOUSE_DOC_DIR "share/doc/stable-mouse")
endif()
install(FILES LICENSE README.md THIRD_PARTY.md DESTINATION ${STABLE_MOUSE_DOC_DIR})
install(DIRECTORY LICENSES docs DESTINATION ${STABLE_MOUSE_DOC_DIR})
if(WIN32 OR APPLE)
  if(Qt6_VERSION VERSION_LESS 6.5)
    qt_generate_deploy_app_script(TARGET stable-mouse FILENAME_VARIABLE deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  else()
    qt_generate_deploy_app_script(TARGET stable-mouse OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  endif()
  install(SCRIPT ${deploy_script})
endif()
if(WIN32)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
  set(CMAKE_INSTALL_UCRT_LIBRARIES TRUE)
  include(InstallRequiredSystemLibraries)
endif()
if(BUILD_TESTING)
  qt_add_executable(app_tests tests/app_tests.cpp src/startup.cpp src/practice.cpp src/practice.h src/window.cpp src/window.h src/backend.h ${STABLE_MOUSE_UI_RESOURCES})
  target_link_libraries(app_tests PRIVATE stabilizer stable_mouse_updater Qt6::Test)
  qt_add_executable(updater_tests tests/updater_tests.cpp)
  set_target_properties(updater_tests PROPERTIES WIN32_EXECUTABLE FALSE MACOSX_BUNDLE FALSE)
  target_link_libraries(updater_tests PRIVATE stable_mouse_updater Qt6::Test)
  add_test(NAME updater COMMAND updater_tests)
  add_test(NAME updater_live COMMAND updater_tests livePublishedRelease -o -,txt)
  set_tests_properties(updater_live PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 180)
  set_tests_properties(updater PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME app COMMAND app_tests)
  set_tests_properties(app PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME smoke COMMAND stable-mouse --smoke-test)
  set_tests_properties(smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  if(WIN32)
    qt_add_executable(windows_backend_tests tests/windows_backend_tests.cpp src/platform/windows.cpp src/backend.h)
    target_link_libraries(windows_backend_tests PRIVATE stabilizer Qt6::Widgets user32)
    target_compile_definitions(windows_backend_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE STABLE_MOUSE_TEST_REPLAY)
    add_executable(windows_focus_tests tests/windows_focus_tests.cpp)
    target_compile_definitions(windows_focus_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
    target_link_libraries(windows_focus_tests PRIVATE user32 ole32)
    add_test(NAME windows_focus COMMAND windows_focus_tests)
    set_tests_properties(windows_focus PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 15)
    add_test(NAME windows_backend COMMAND windows_backend_tests)
    set_tests_properties(windows_backend PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 90)
    add_test(NAME windows_backend_always COMMAND windows_backend_tests)
    set_tests_properties(windows_backend_always PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 90 ENVIRONMENT "STABLE_MOUSE_TEST_NATIVE_INPUT=1;STABLE_MOUSE_TEST_CENTER=1;STABLE_MOUSE_TEST_ALWAYS=1")
  endif()
endif()
set(CPACK_PACKAGE_NAME "stable-mouse")
set(CPACK_VERBATIM_VARIABLES TRUE)
set(CPACK_PACKAGE_VENDOR "Stable Mouse contributors")
set(CPACK_PACKAGE_CONTACT "Stable Mouse contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Adjustable mouse stabilization")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
if(WIN32)
  set(CPACK_GENERATOR NSIS)
  if(CMAKE_SIZEOF_VOID_P EQUAL 8 AND NOT CMAKE_GENERATOR_PLATFORM MATCHES "[Aa][Rr][Mm]" AND NOT CMAKE_SYSTEM_PROCESSOR MATCHES "[Aa][Rr][Mm]|aarch64")
    set(CPACK_PACKAGE_FILE_NAME "Stable-Mouse-${PROJECT_VERSION}-Windows-x64")
  endif()
  set(CPACK_NSIS_DISPLAY_NAME "Stable Mouse")
  set(CPACK_NSIS_PACKAGE_NAME "Stable Mouse")
  set(CPACK_NSIS_MUI_ICON "${CMAKE_CURRENT_SOURCE_DIR}/packaging/stable-mouse.ico")
  set(CPACK_NSIS_MUI_UNIICON "${CMAKE_CURRENT_SOURCE_DIR}/packaging/stable-mouse.ico")
  set(CPACK_NSIS_INSTALLED_ICON_NAME "bin\\\\stable-mouse.exe")
  set(CPACK_PACKAGE_INSTALL_DIRECTORY "Stable Mouse")
  set(CPACK_NSIS_EXECUTABLES_DIRECTORY bin)
  set(CPACK_PACKAGE_EXECUTABLES "stable-mouse" "Stable Mouse")
  set(CPACK_NSIS_EXTRA_UNINSTALL_COMMANDS "DeleteRegValue HKCU \"Software\\Microsoft\\Windows\\CurrentVersion\\Run\" \"StableMouse\"")
elseif(APPLE)
  set(CPACK_GENERATOR DragNDrop)
  # The release workflow builds both architectures in this image.
  if("${CMAKE_OSX_ARCHITECTURES}" MATCHES "arm64" AND "${CMAKE_OSX_ARCHITECTURES}" MATCHES "x86_64")
    set(CPACK_PACKAGE_FILE_NAME "Stable-Mouse-${PROJECT_VERSION}-macOS-universal")
  endif()
  set(CPACK_DMG_SLA_USE_RESOURCE_FILE_LICENSE OFF)
else()
  set(CPACK_GENERATOR DEB)
  # Identify the build distribution; a generic Linux package may not run on it.
  cmake_host_system_information(RESULT STABLE_MOUSE_OS QUERY DISTRIB_ID)
  cmake_host_system_information(RESULT STABLE_MOUSE_OS_VERSION QUERY DISTRIB_VERSION_ID)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    if(STABLE_MOUSE_OS STREQUAL "debian" AND STABLE_MOUSE_OS_VERSION STREQUAL "12")
      set(CPACK_PACKAGE_FILE_NAME "stable-mouse-${PROJECT_VERSION}-Debian12-amd64")
    elseif(STABLE_MOUSE_OS STREQUAL "ubuntu" AND STABLE_MOUSE_OS_VERSION STREQUAL "24.04")
      set(CPACK_PACKAGE_FILE_NAME "stable-mouse-${PROJECT_VERSION}-Ubuntu24.04-amd64")
    endif()
  endif()
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  set(CPACK_DEBIAN_PACKAGE_SECTION "utils")
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "pkexec, libqt6svg6, libqt6gui6, qt6-qpa-plugins")
  set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
endif()
include(CPack)
